\subsection*{Character states}
\begin{description}\setlength{\itemsep}{2pt}
\item[\texttt{CN1}] \textit{contribution timing} — \texttt{0}~=~ex-ante-periodic; \texttt{1}~=~ex-ante-lump; \texttt{2}~=~ex-post-assessment
\item[\texttt{HZ1}] \textit{hazard correlation} — \texttt{0}~=~idiosyncratic; \texttt{1}~=~covariate; \texttt{2}~=~mixed
\item[\texttt{HZ2}] \textit{hazard responsiveness} — ternary: \texttt{1} present, \texttt{P} partial, \texttt{0} absent
\item[\texttt{MB3}] \textit{participation basis} — \texttt{0}~=~voluntary; \texttt{1}~=~compulsory
\item[\texttt{MC1}] \textit{mitigation mechanism} — \texttt{0}~=~allocation; \texttt{1}~=~spreading; \texttt{2}~=~pooling
\item[\texttt{PR1}] \textit{peril priced ex ante} — ternary: \texttt{1} present, \texttt{P} partial, \texttt{0} absent
\item[\texttt{PY0}] \textit{pool output} — \texttt{0}~=~collective-good; \texttt{1}~=~individual-indemnity; \texttt{2}~=~individual-draw
\item[\texttt{PY1}] \textit{payout trigger} — \texttt{0}~=~realised-loss; \texttt{1}~=~life-event; \texttt{2}~=~rotation; \texttt{3}~=~need-assessed
\item[\texttt{PY2}] \textit{allocation rule} — \texttt{0}~=~rotation-lot; \texttt{1}~=~rotation-fixed; \texttt{2}~=~auction; \texttt{3}~=~indemnity; \texttt{4}~=~need-assessed
\end{description}
